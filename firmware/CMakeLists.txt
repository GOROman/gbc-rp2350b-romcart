cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD romcart)
set(PICO_BOARD_HEADER_DIRS ${CMAKE_CURRENT_LIST_DIR}/boards)
set(PICO_PLATFORM rp2350-arm-s)
if(NOT DEFINED ENV{PICO_SDK_PATH})
 message(FATAL_ERROR "Set PICO_SDK_PATH to an existing pico-sdk; this project does not install dependencies")
endif()
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(romcart C CXX ASM)
pico_sdk_init()
add_executable(romcart main.c)
pico_generate_pio_header(romcart ${CMAKE_CURRENT_LIST_DIR}/rom_bus.pio)
target_link_libraries(romcart pico_stdlib hardware_pio hardware_sync)
pico_enable_stdio_usb(romcart 0)
pico_enable_stdio_uart(romcart 0)
pico_add_extra_outputs(romcart)
