cmake_minimum_required(VERSION 3.13)
set(PICO_PLATFORM rp2350)
set(PICO_BOARD gbc_rom)
set(PICO_BOARD_HEADER_DIRS ${CMAKE_CURRENT_LIST_DIR}/boards)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(gbc_rom C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(gbc_rom main.c)
pico_generate_pio_header(gbc_rom ${CMAKE_CURRENT_LIST_DIR}/rom_bus.pio)
target_link_libraries(gbc_rom pico_stdlib hardware_pio hardware_dma hardware_clocks hardware_sync)
pico_enable_stdio_usb(gbc_rom 0)
pico_enable_stdio_uart(gbc_rom 0)
pico_add_extra_outputs(gbc_rom)
